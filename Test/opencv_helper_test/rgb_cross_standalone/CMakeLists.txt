cmake_minimum_required(VERSION 3.20)
project(RgbCrossMeasurement LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_executable(rgb_cross_cli "${REPO}/Test/opencv_helper_test/rgb_cross_cli.cpp" "${REPO}/Native/opencv_helper/algorithm/find_cross/rgb_cross.cpp")
target_include_directories(rgb_cross_cli PRIVATE "${REPO}/Native/opencv_helper/algorithm/find_cross" "${REPO}/packages/nlohmann/single_include")
if(MSVC)
  target_compile_options(rgb_cross_cli PRIVATE /utf-8 /fp:precise)
endif()
enable_testing()
add_test(NAME rgb_cross_native COMMAND rgb_cross_cli --self-test)
